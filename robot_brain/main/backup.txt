extern "C" void app_main(void)
{
    Display display;
    Servo neck_pan;
    
    // =====================================================
    // 1. Initialize Display Matrix Engine
    // =====================================================
    esp_err_t display_err = display.init();
    if (display_err != ESP_OK) {
        ESP_LOGE("MAIN", "Display initialization failed: %s", esp_err_to_name(display_err));
        return;
    } else {
        ESP_LOGI("MAIN", "Display initialized successfully");
    }

    // Paint initial resting background layout face looking forward
    display.draw_default_eyes(0, 0);

    // =====================================================
    // 2. Initialize Servo Hardware PWM Channels
    // =====================================================
    // Maps your physical servo signal pin to GPIO 5
    esp_err_t servo_err = neck_pan.init(5, LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0);
    if (servo_err != ESP_OK) {
        ESP_LOGE("MAIN", "Servo initialization failed: %s", esp_err_to_name(servo_err));
        return;
    } else {
        ESP_LOGI("MAIN", "Servo hardware linked to GPIO 5 successfully");
    }

    // Pause briefly to let voltages stabilize before launching tracking routines
    vTaskDelay(pdMS_TO_TICKS(1000));

    // =====================================================
    // 3. Multitasking Animation Expression Loop
    // =====================================================
    while (true) {
        // Step A: Look Right on screen AND pan physical robot neck right
        ESP_LOGI("ROBOT", "Expression Action: Tracking RIGHT...");
        neck_pan.set_angle(135, 1000, Servo::Direction::CLOCKWISE);
        display.animate_pupil_move(20, 0, 400); 
        vTaskDelay(pdMS_TO_TICKS(2000));

        // Step B: Look Left on screen AND pan physical robot neck left
        ESP_LOGI("ROBOT", "Expression Action: Tracking LEFT...");
        neck_pan.set_angle(45, 1500, Servo::Direction::CLOCKWISE);
        display.animate_pupil_move(-20, 0, 400);
        vTaskDelay(pdMS_TO_TICKS(2000));

        // Step C: Blink eyes and bring everything back to center home-base axis
        ESP_LOGI("ROBOT", "Expression Action: Re-centering & Blinking...");
        neck_pan.middle();
        display.animate_pupil_move(0, 0, 300);
        display.animate_blink(120);
        vTaskDelay(pdMS_TO_TICKS(4000));
    }
}

void test_speaker(Sound &sound)
{
    // sound.enable_speakers();

    constexpr float LOW_FREQ = 500.0f;
    constexpr float HIGH_FREQ = 1500.0f;
    constexpr float SWEEP_PERIOD = 1.0f;
    constexpr float AMPLITUDE = 3000000.0f; // Perfect for 24-bit audio depth
    constexpr float FADE_SECONDS = 0.02f;
    constexpr float PI = 3.14159265358979323846f;

    constexpr size_t DURATION_SECONDS = 5;
    constexpr size_t CHUNK_FRAMES = 512;

    const size_t total_frames = Sound::SAMPLE_RATE * DURATION_SECONDS;
    const float fade_samples = FADE_SECONDS * Sound::SAMPLE_RATE;

    std::vector<uint8_t> buffer(CHUNK_FRAMES * 2 * 3);
    // std::vector<int32_t> buffer(CHUNK_FRAMES * 2);

    std::vector<int32_t> dump(CHUNK_FRAMES);

    float oscillator_phase = 0.0f;
    size_t generated = 0;

    while (generated < total_frames)
    {
        const size_t frames = std::min(CHUNK_FRAMES, total_frames - generated);

        for (size_t i = 0; i < frames; ++i)
        {
            const size_t sample_index = generated + i;
            const float t = static_cast<float>(sample_index) / Sound::SAMPLE_RATE;

            // 1. Calculate the target frequency for this exact moment
            const float sweep = 0.5f * (1.0f + std::sinf(2.0f * PI * t / SWEEP_PERIOD));
            const float frequency = LOW_FREQ + sweep * (HIGH_FREQ - LOW_FREQ);

            // 2. Continuous phase step (No fmodf for speed)
            oscillator_phase += (2.0f * PI * frequency) / static_cast<float>(Sound::SAMPLE_RATE);
            if (oscillator_phase >= 2.0f * PI)
            {
                oscillator_phase -= 2.0f * PI;
            }

            // 3. Smooth Envelope Fading
            float envelope = 1.0f;
            if (sample_index < fade_samples) // Fade-in
            {
                envelope = static_cast<float>(sample_index) / fade_samples;
            }
            else if ((total_frames - sample_index) < fade_samples) // Fade-out
            {
                envelope = static_cast<float>(total_frames - sample_index) / fade_samples;
            }

            // 4. Generate audio value
            const float value = std::sinf(oscillator_phase) * AMPLITUDE * envelope;
            const int32_t sample = static_cast<int32_t>(value);

            // Keep only the 24-bit two's-complement representation.
            const uint32_t packed =
                static_cast<uint32_t>(sample) &
                0x00FFFFFF;

            // -------------------------------------------------
            // One stereo frame = 6 bytes
            //
            // [ L2 L1 L0 ][ R2 R1 R0 ]
            // -------------------------------------------------

            const size_t offset =
                i * 6;

            // LEFT CHANNEL = silence
            buffer[offset + 0] =
                static_cast<uint8_t>(
                    (packed >> 16) & 0xFF);
            // RIGHT CHANNEL = siren
            buffer[offset + 1] =
                static_cast<uint8_t>(
                    (packed >> 8) & 0xFF);

            buffer[offset + 2] = static_cast<uint8_t>(
                packed & 0xFF);

            buffer[offset + 3] = 0;

            buffer[offset + 4] =
                0;

            buffer[offset + 5] = 0;
        }

        // const esp_err_t err = sound.speak(buffer.data(), buffer.size(), CHUNK_FRAMES * 2);

        const esp_err_t err = sound.speak_r(buffer.data(), buffer.size(), CHUNK_FRAMES * 2);
        if (err != ESP_OK)
        {
            ESP_LOGE("SirenTest", "speak() failed: %s", esp_err_to_name(err));
            return;
        }

        generated += frames;
    }

    ESP_LOGW("SirenTest", "Siren finished.");
}
